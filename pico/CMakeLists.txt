cmake_minimum_required(VERSION 3.17)
set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../deps/pico-sdk" CACHE PATH "Pico SDK")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(wireless_uart_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(PICO_NO_PICOTOOL 1)
pico_sdk_init()
add_executable(wireless_uart_pico main.c bridge_engine.c usb_descriptors.c)
target_include_directories(wireless_uart_pico PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/../firmware)
target_link_libraries(wireless_uart_pico pico_stdlib hardware_uart hardware_irq pico_unique_id tinyusb_device tinyusb_board)
target_compile_options(wireless_uart_pico PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_usb(wireless_uart_pico 0)
pico_enable_stdio_uart(wireless_uart_pico 0)
pico_add_extra_outputs(wireless_uart_pico)
